   75: def _cosine_ramp(n):
   76:     # rampa coseno de 0->1 en n muestras (n>=1)
   77:     if n <= 0:
   78:         return np.array([], dtype=np.float64)
   79:     t = np.linspace(0, np.pi, n, endpoint=True)
   80:     return 0.5 - 0.5*np.cos(t)  # 0..1

   82: def _feather_mask_from_core(h, w, core, margin_y, margin_x):
   83:     """
   84:     Crea una máscara 2D con plateau=1 en 'core' (en coords del parche)
   85:     y rampas coseno en los márgenes hacia fuera (hasta 'margin_*').
   86:     Fuera de core + rampas -> 0.
   87:     """
   88:     y0, y1, x0, x1 = core
   89:     y0 = max(0, min(h, y0)); y1 = max(0, min(h, y1))
   90:     x0 = max(0, min(w, x0)); x1 = max(0, min(w, x1))
   91: 
   92:     wy = np.zeros(h, dtype=np.float64)
   93:     wx = np.zeros(w, dtype=np.float64)
   94: 
   95:     # Eje Y
   96:     mY_top = min(margin_y, y0)              # espacio para rampa arriba
   97:     mY_bot = min(margin_y, h - y1)          # espacio para rampa abajo
   98:     if y0 - mY_top < y1 + mY_bot:
   99:         # tramo superior: 0 -> 1
  100:         if mY_top > 0:
  101:             wy[y0 - mY_top:y0] = _cosine_ramp(mY_top)
  102:         # plateau
  103:         wy[y0:y1] = 1.0
  104:         # tramo inferior: 1 -> 0
  105:         if mY_bot > 0:
  106:             wy[y1:y1 + mY_bot] = _cosine_ramp(mY_bot)[::-1]
  107: 
  108:     # Eje X
  109:     mX_left  = min(margin_x, x0)
  110:     mX_right = min(margin_x, w - x1)
  111:     if x0 - mX_left < x1 + mX_right:
  112:         if mX_left > 0:
  113:             wx[x0 - mX_left:x0] = _cosine_ramp(mX_left)
  114:         wx[x0:x1] = 1.0
  115:         if mX_right > 0:
  116:             wx[x1:x1 + mX_right] = _cosine_ramp(mX_right)[::-1]
  117: 
  118:     mask = np.outer(wy, wx)
  119:     return mask

  121: def _cosine_taper_2d(h, w, margin_y, margin_x):
  122:     # Hann 2D sobre los bordes del parche (pre-apodización, pequeña)
  123:     my = int(max(0, margin_y)); mx = int(max(0, margin_x))
  124:     wy = np.ones(h, dtype=np.float64)
  125:     wx = np.ones(w, dtype=np.float64)
  126:     if my > 0:
  127:         r = _cosine_ramp(my)
  128:         wy[:my] = r
  129:         wy[-my:] = r[::-1]
  130:     if mx > 0:
  131:         r = _cosine_ramp(mx)
  132:         wx[:mx] = r
  133:         wx[-mx:] = r[::-1]
  134:     return np.outer(wy, wx)

  136: def apply_quadrant_shifts_with_feather(
  137:     img2d,                       # imagen 2D (X,Y)
  138:     quadrants_roi,               # lista de (y1,y2,x1,x2)
  139:     shifts_q,                    # lista de (dy,dx) por cuadrante (en mismo orden)
  140:     margin=10,                   # px de margen para feather (solape)
  141:     shift_func=None,             # callable(patch, shift=[dy,dx], wrap=False, fill=0)
  142:     wrap=False, fill=0,
  143:     pre_apod=3                   # apodización pequeña pre-shift (px)
  144: ):
  145:     """
  146:     - Expande cada ROI en margin + |shift| (por eje) para garantizar cobertura tras el desplazamiento.
  147:     - Aplica ventana coseno suave en los bordes del parche (pre_apod) antes del shift (reduce ringing).
  148:     - Construye una máscara de escritura con plateau=1 exactamente en el ROI original y rampas en el solape.
  149:     - Mezcla y normaliza (suma ponderada); partición de unidad en solapes -> sin “cruz” en el centro.
  150:     """
  151:     H, W = img2d.shape
  152:     num = np.zeros_like(img2d, dtype=np.float64)
  153:     den = np.zeros_like(img2d, dtype=np.float64)
  154: 
  155:     for (y1, y2, x1, x2), (dy, dx) in zip(quadrants_roi, shifts_q):
  156:         # expansión dependiente del shift (garantiza cobertura post-shift)
  157:         pad_y = margin + int(np.ceil(abs(dy)))
  158:         pad_x = margin + int(np.ceil(abs(dx)))
  159: 
  160:         ye1 = max(0, y1 - pad_y)
  161:         ye2 = min(H, y2 + pad_y)
  162:         xe1 = max(0, x1 - pad_x)
  163:         xe2 = min(W, x2 + pad_x)
  164:         if ye2 <= ye1 or xe2 <= xe1:
  165:             continue
  166: 
  167:         patch = img2d[ye1:ye2, xe1:xe2]
    → 'patch' es VISTA (slicing/indexación)
  168:         h, w = patch.shape
  169: 
  170:         # 1) Apodización pre-shift (pequeña, sobre bordes del parche)
  171:         apy = min(pre_apod, h//2 - 1) if h >= 3 else 0
  172:         apx = min(pre_apod, w//2 - 1) if w >= 3 else 0
  173:         pre_w = _cosine_taper_2d(h, w, apy, apx)
  174:         patch_win = patch * pre_w
  175: 
  176:         # 2) Desplaza el parche
  177:         shifted = shift_func(patch_win, shift=[dy, dx], wrap=wrap, fill=fill)
  178: 
  179:         # 3) Máscara de escritura basada en el núcleo (ROI original)
  180:         core_y0 = y1 - ye1
  181:         core_y1 = y2 - ye1
  182:         core_x0 = x1 - xe1
  183:         core_x1 = x2 - xe1
  184: 
  185:         # márgenes efectivos (no más grandes que el espacio disponible)
  186:         my = min(margin, h//2)
  187:         mx = min(margin, w//2)
  188:         write_w = _feather_mask_from_core(
  189:             h, w,
  190:             core=(core_y0, core_y1, core_x0, core_x1),
  191:             margin_y=my, margin_x=mx
  192:         )
  193: 
  194:         # 4) Acumulación
  195:         num[ye1:ye2, xe1:xe2] += shifted * write_w
  196:         den[ye1:ye2, xe1:xe2] += write_w
  197: 
  198:     out = img2d.copy()
  199:     m = den > 1e-12
  200:     out[m] = num[m] / den[m]
  201:     return out

  203: def apply_quadrant_shifts_with_feather_stack(
  204:     stack4, quadrants_roi, shifts_q_per_mod, margin=10, shift_func=None, wrap=False, fill=0
  205: ):
  206:     """
  207:     Aplica shifts por cuadrante a un stack (4,X,Y).
  208:     shifts_q_per_mod: lista (por mod j) de listas (por cuadrante q) de tuples (dy,dx),
  209:                       es decir: longitud 4, cada elemento longitud Q.
  210:     """
  211:     out = stack4.copy()
  212:     nmods, H, W = stack4.shape
  213:     assert len(shifts_q_per_mod) == nmods, "Se espera una lista de shifts por mod (len=4)."
  214: 
  215:     for j in range(nmods):
  216:         shifts_q = shifts_q_per_mod[j]  # lista de (dy,dx) por cuadrante
    → 'shifts_q' es VISTA (slicing/indexación)
  217:         out[j] = apply_quadrant_shifts_with_feather(
  218:             out[j], quadrants_roi, shifts_q, margin=margin,
  219:             shift_func=shift_func, wrap=wrap, fill=fill
  220:         )
  221:     return out

  442: def compute_pseudo_images_leastsq(imgs_cam1, imgs_cam2, M1, M2, roi=None, eps=1e-12):
  443:     """
  444:     Construye pseudo‑imágenes 'tipo I' por modulación j, combinando CAM1 y CAM2
  445:     con pesos α,β que:
  446:        (i) preservan I  (α+β = 1)
  447:       (ii) minimizan la fuga de QUV  (|| α m1 + β m2 ||)
  448: 
  449:     Además aplica una normalización fotométrica por estado (opcional) para
  450:     equiparar las escalas de ambas cámaras antes de combinar.
  451: 
  452:     Parámetros
  453:     ----------
  454:     imgs_cam1, imgs_cam2 : (4, X, Y)
  455:         Imágenes moduladas, ya alineadas entre cámaras.
  456:     M1, M2 : (4,4)
  457:         Matrices de modulación de cam1 y cam2. Cada fila j: [1, a, b, c].
  458:     roi : tuple o None
  459:         (x0, x1, y0, y1) para estimar la normalización fotométrica por estado.
  460:         Si None, usa toda la imagen con percentiles robustos.
  461:     eps : float
  462:         Tolerancia numérica para evitar divisiones por cero.
  463: 
  464:     Devuelve
  465:     --------
  466:     pseudo : (4, X, Y)
  467:         Pseudo‑imágenes por modulación (aprox. 'I').
  468:     alphas, betas : (4,)
  469:         Pesos por estado j.
  470:     gammas : (4,)
  471:         Factores de normalización aplicados a cam2 por estado j.
  472:     residual_norms : (4,)
  473:         Norma de la fuga residual || α m1 + β m2 || por estado j.
  474:     """
  475:     nmods, X, Y = imgs_cam1.shape
  476:     pseudo = np.zeros_like(imgs_cam1, dtype=np.float64)
  477:     alphas = np.zeros(nmods)
  478:     betas  = np.zeros(nmods)
  479:     gammas = np.ones(nmods)
  480:     residual_norms = np.zeros(nmods)
  481: 
  482:     for j in range(nmods):
  483:         I1 = imgs_cam1[j].astype(np.float64)
  484:         I2 = imgs_cam2[j].astype(np.float64)
  485: 
  486:         # --- Normalización fotométrica por estado j (cam2 -> cam1) ---
  487:         if roi is not None:
  488:             x0, x1, y0, y1 = roi
  489:             ref1 = I1[x0:x1, y0:y1]
    → 'ref1' es VISTA (slicing/indexación)
  490:             ref2 = I2[x0:x1, y0:y1]
    → 'ref2' es VISTA (slicing/indexación)
  491:         else:
  492:             # Percentiles robustos en toda la imagen
  493:             ref1 = I1
  494:             ref2 = I2
  495: 
  496:         # Evitar outliers (usar median de ratios donde I1 no es casi cero)
  497:         mask = np.abs(ref1) > np.percentile(np.abs(ref1), 5) + eps
  498:         if np.any(mask):
  499:             ratio = ref2[mask] / (ref1[mask] + eps)
  500:             gamma = np.median(ratio)
  501:         else:
  502:             gamma = 1.0
  503: 
  504:         I2n = I2 / (gamma + eps)
  505:         gammas[j] = gamma
  506: 
  507:         # --- Pesos óptimos por mínimos cuadrados con α+β=1 ---
  508:         m1 = M1[j, 1:].astype(np.float64)  # (a,b,c)
  509:         m2 = M2[j, 1:].astype(np.float64)  # (a',b',c')
  510: 
  511:         diff = m1 - m2
  512:         denom = np.dot(diff, diff)
  513:         if denom < eps:
  514:             # Caso degenerado: m1 ≈ m2 → cualquier α con α+β=1 sirve; usa α=β=0.5
  515:             alpha = 0.5
  516:         else:
  517:             alpha = (np.dot(m2, m2) - np.dot(m1, m2)) / denom
  518:         beta = 1.0 - alpha
  519: 
  520:         alphas[j] = alpha
  521:         betas[j]  = beta
  522: 
  523:         # Fuga residual teórica (diagnóstico)
  524:         residual = alpha * m1 + beta * m2
  525:         residual_norms[j] = np.linalg.norm(residual)
  526: 
  527:         # --- Pseudo‑imagen ---
  528:         pseudo[j] = alpha * I1 + beta * I2n
  529: 
  530:         # (Opcional) Re‑escala para visualizar comparables
  531:         # pseudo[j] *= 1.0  # mantener unidad de I por construcción (α+β=1)
  532: 
  533:     return pseudo, alphas, betas, gammas, residual_norms

  629: def dftreg(F,G,kappa):
  630:     """
  631:     Calculates the shift between a couple of images 'f' and 'g' with subpixel
  632:     accuracy following the second method presented in
  633:     Sicairos 2008, Efficient subpixel image registration algorithm.
  634:     Input:
  635:         F,G: ffts of images 'f' and 'g' without applying any fftshift
  636:         kappa: inverse of subpixel precision (kappa=20 -> 0.05 pixel precision)
  637:     Output:
  638: 
  639:     """
  640:     nr,nc=np.shape(F)
  641:     Nr = ifftshift(np.arange(-np.fix(nr/2),np.ceil(nr/2)))
  642:     Nc = ifftshift(np.arange(-np.fix(nc/2),np.ceil(nc/2)))
  643:     Nout = 2 * max(nr, nc)
  644:     CC=ifft2(FTpad(F*np.conj(G), Nout))
  645:     CCabs=np.abs(CC)
  646:     ind = np.unravel_index(np.argmax(CCabs, axis=None), CCabs.shape)
  647: 
  648:     CCmax=CC[ind]*nr*nc
  649:     Nr2 = ifftshift(np.arange(-np.fix(nr),np.ceil(nr)))
  650:     Nc2 = ifftshift(np.arange(-np.fix(nc),np.ceil(nc)))
  651: 
  652:     row_shift=Nr2[ind[0]]/2
  653:     col_shift=Nc2[ind[1]]/2
  654: 
  655:     #Initial shift estimate in upsampled grid
  656:     row_shift=round(row_shift*kappa)/kappa
  657:     col_shift=round(col_shift*kappa)/kappa
  658:     dftshift=np.fix(np.ceil(kappa*1.5)/2)
  659: 
  660:     #DFT by matrix multiplication
  661:     CC=np.conj(dftups(G*np.conj(F),np.ceil(kappa*1.5),kappa,\
  662:     dftshift-row_shift*kappa,dftshift-col_shift*kappa))
  663:     CCabs=np.abs(CC)
  664:     ind = np.unravel_index(np.argmax(CCabs, axis=None), CCabs.shape)
  665:     CCmax=CC[ind]
    → 'CCmax' es VISTA (slicing/indexación)
  666:     rloc,cloc=ind-dftshift
  667:     row_shift=row_shift+rloc/kappa
  668:     col_shift=col_shift+cloc/kappa
  669:     rg00=np.sum(np.abs(F)**2)
  670:     rf00=np.sum(np.abs(G)**2)
  671:     error=np.sqrt(1-np.abs(CCmax)**2/(rg00*rf00))
  672:     Nc,Nr=np.meshgrid(Nc,Nr)
  673:     Gshift=G*np.exp(1j*2*np.pi*(-row_shift*Nr/nr-col_shift*Nc/nc))
  674:     return error,row_shift,col_shift,Gshift

  676: def dftups(M,n_out,kappa,roff,coff):
  677:     """
  678:     Upsampled cross-correlation obtained by matrix multiplication
  679:     Inputs:
  680:         M: input image for calculation of the DFT
  681:         n_out: number of pixels in the output upsampled DFT
  682:         kappa: inverse of subpixel precision (kappa=20 -> 0.005 pixel precision)
  683:         roff, coff: row and column offsets to shift the output array to a
  684:             region of interest
  685:     """
  686:     nr,nc=M.shape
  687:     kernc=np.exp((-1j*2*np.pi/(nc*kappa))*np.outer(\
  688:     ifftshift(np.arange(0,nc).T-np.floor(nc/2)),np.arange(0,n_out)-coff))
  689: 
  690:     kernr=np.exp((-1j*2*np.pi/(nr*kappa))*np.outer(\
  691:     np.arange(0,n_out)-roff,ifftshift(np.arange(0,nr).T-np.floor(nr/2))))
  692:     return kernr @ M @ kernc

  694: def FTpad(IM,Nout):
  695:     """
  696:     Carries out zero-padding to upsample an image IM in Fourier domain
  697:     Input:
  698:         IM: Numpy array in Fourier domain
  699:         outsize: size of the new array
  700: 
  701:     """
  702:     Nin=IM.shape[0]
    → 'Nin' es VISTA (slicing/indexación)
  703:     pd=int((Nout-Nin)/2)
  704:     IM=fftshift(IM)
  705:     IMout=np.pad(IM,((pd,pd),(pd,pd)),'constant')
  706:     IMout=ifftshift(IMout)*Nout*Nout/(Nin*Nin)
  707:     return IMout

  828:         Input is y and x shifts (defined negative towards (0,0)

  833:     nr, nc = im.shape
  834:     Nr = ifftshift(np.arange(-np.fix(nr / 2), np.ceil(nr / 2)))

  909:     Se calculan (I, Q, U, V) de cada cámara por separado sin intentar alinear
  910:     aún las modulaciones crudas.
